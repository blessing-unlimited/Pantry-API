FROM --platform=linux/amd64 mcr.microsoft.com/dotnet/sdk:10.0 AS build

WORKDIR /src
COPY . .

RUN dotnet restore Pantry.slnx
RUN dotnet publish Pantry.Api/Pantry.Api.csproj \
    --configuration Release \
    --output /app/publish \
    /p:UseAppHost=false

FROM --platform=linux/amd64 mcr.microsoft.com/dotnet/aspnet:10.0 AS runtime

WORKDIR /app
COPY --from=build /app/publish .

ENV ASPNETCORE_ENVIRONMENT=Production

EXPOSE 10000

ENTRYPOINT ["sh", "-c", "dotnet Pantry.Api.dll --urls http://0.0.0.0:${PORT:-10000}"]
